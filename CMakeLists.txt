cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "")
project(osureplayer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_link_options(/DEBUG:NONE)
endif()

add_library(replay_core replay.cpp playback.cpp game_sync.cpp beatmap.cpp third_party/lzma/LzmaDec.c)
target_include_directories(replay_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(osureplayer main.cpp windows_player.cpp stable_memory.cpp console_io.cpp)
target_link_libraries(osureplayer PRIVATE replay_core user32 winmm)
if(MSVC)
    target_compile_options(replay_core PRIVATE /W4 /utf-8)
    target_compile_options(osureplayer PRIVATE /W4 /utf-8)
endif()
